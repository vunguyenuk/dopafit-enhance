struct ExercisePicker:View {
    @EnvironmentObject var store:AppStore
    @Environment(\.dismiss) var dismiss
    var title:String
    var replacement=false
    var replacing:Exercise?=nil
    var onChoose:(Exercise)->Void
    @State private var query=""
    @State private var gear="Tất cả"
    @State private var sort="Hợp nhất"
    var choices:[Exercise] {
        store.exercises(query:query,gear:gear,sort:sort,replacing:replacing).filter{!replacement || $0.id != replacing?.id}
    }
    var body:some View {
        VStack(spacing:12) {
            SheetTitle(title:title)
            HStack {Image(systemName:"magnifyingglass").foregroundStyle(Dopa.muted);TextField("Tìm bài tập",text:$query);if !query.isEmpty {Button{query=""}label:{Image(systemName:"xmark.circle.fill")}}}.padding(13).background(Dopa.card).clipShape(Capsule()).padding(.horizontal,16)
            HStack {
                Menu {Picker("Sắp xếp",selection:$sort){ForEach(["Hợp nhất","Tên bài tập","Tập nhiều nhất","Tập ít nhất","Chưa tập"],id:\.self){Text($0)}}}label:{Label(sort,systemImage:"arrow.up.arrow.down")}
                Spacer()
                Menu {Picker("Dụng cụ",selection:$gear){ForEach(["Tất cả"]+Catalog.equipment,id:\.self){Text($0)}}}label:{Label(gear,systemImage:"line.3.horizontal.decrease")}
            }.font(.system(size:15)).foregroundStyle(.white).padding(.horizontal,20)
            ScrollView {
                LazyVStack(spacing:18) {
                    ForEach(choices) { ex in
                        Button { onChoose(ex); if !replacement {dismiss()} } label: {
                            HStack(spacing:16) {ExerciseThumb(exercise:ex,size:66);VStack(alignment:.leading,spacing:5){Text(store.english ? ex.english:ex.name).font(.system(size:17,weight:.medium)).foregroundStyle(.white);Text(ex.equipment).font(.system(size:14)).foregroundStyle(Dopa.muted)};Spacer()}
                        }.buttonStyle(.plain)
                    }
                    if choices.isEmpty {Text("Không tìm thấy bài tập").foregroundStyle(Dopa.muted).padding(40)}
                }.padding(16)
            }
        }.background(.black).presentationDragIndicator(.visible)
    }
}
