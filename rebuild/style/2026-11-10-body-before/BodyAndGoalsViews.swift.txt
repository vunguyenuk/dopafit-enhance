struct RecoveryMusclesView:View {
    @EnvironmentObject var store:AppStore
    @Environment(\.dismiss) var dismiss
    @State private var selected:String?=nil
    @State private var editing=false
    var body:some View {
        VStack(spacing:18) {
            HStack{CircleButton(icon:"chevron.left"){dismiss()};Spacer();Text("Hồi phục nhóm cơ").font(.system(size:19,weight:.semibold));Spacer();Button("Sửa"){editing=true}.foregroundStyle(Dopa.lime).frame(width:52,height:44).background(Dopa.card).clipShape(Capsule())}.padding(.horizontal,16)
            ScrollView {
                VStack(alignment:.leading,spacing:24) {
                    recoveryGroup("Nhóm cơ chính",muscles:Catalog.recoveryMuscles)
                    recoveryGroup("Nhóm cơ phụ",muscles:Catalog.recoverySecondaryMuscles)
                }.padding(.horizontal,16).padding(.bottom,25)
            }
        }.background(Dopa.background).toolbar(.hidden,for:.navigationBar)
            .navigationDestination(isPresented:$editing){RecoveryEditor()}
            .sheet(isPresented:Binding(get:{selected != nil},set:{if !$0{selected=nil}})) {if let muscle=selected {RecoveryMuscleDetail(muscle:muscle)}}
    }
    func recoveryGroup(_ title:String,muscles:[String])->some View {
        VStack(alignment:.leading,spacing:18) {
            SectionLabel(title:title)
            GroupCard {ForEach(muscles,id:\.self){muscle in
                Button{selected=muscle}label:{HStack(spacing:14){MuscleThumbnail(muscle:muscle);VStack(alignment:.leading,spacing:6){Text(muscle).font(.system(size:20,weight:.semibold));Text(lastExercise(muscle)).font(.system(size:14)).foregroundStyle(Dopa.muted).lineLimit(2)};Spacer();Text("\(Int(store.muscleRecovery(muscle)))%").font(.system(size:20,weight:.semibold)).foregroundStyle(store.muscleRecovery(muscle)>=100 ? .white:.orange)}.padding(16).foregroundStyle(.white)}.buttonStyle(.plain)
                if muscle != muscles.last {Divider().padding(.horizontal,16)}
            }}
        }
    }
    func lastExercise(_ muscle:String)->String {
        for session in store.data.history.sorted(by:{$0.started>$1.started}) {
            if let item=session.exercises.first(where:{$0.exercise.muscles.contains(muscle)||$0.exercise.secondary.contains(muscle)}) {
                let hours=max(0,Int(Date().timeIntervalSince(session.started)/3600))
                let age=hours==0 ? "dưới 1 giờ trước":"\(hours) giờ trước"
                return item.exercise.name+", "+age
            }
        }
        return "Chưa tập gần đây"
    }
}

                Text("Chỉnh theo cảm giác thực tế của bạn. Bài tập hôm nay sẽ ưu tiên nhóm cơ đã hồi phục và bỏ qua nhóm cơ ở 0%.").foregroundStyle(Dopa.muted).font(.system(size:16))

struct GoalGroup {
    var title:String
    var muscles:[String]
    var targets:[Double]
    var color:Color
    var target:Double {targets.reduce(0,+)}
}
struct GoalsView:View {
    @EnvironmentObject var store:AppStore
    @State private var expanded:Set<Int>=[]
    @State private var help=false
    let groups=[GoalGroup(title:"Cơ nhóm đẩy",muscles:["Ngực","Vai","Tay sau"],targets:[7,12,8],color:Dopa.push),GoalGroup(title:"Cơ nhóm kéo",muscles:["Lưng","Tay trước"],targets:[7,7],color:Dopa.pull),GoalGroup(title:"Cơ chân",muscles:["Đùi trước","Đùi sau","Mông","Bắp chân"],targets:[7,6,7,4],color:Dopa.legs)]
    var total:Double {groups.reduce(0){sum,g in sum+g.muscles.reduce(0){$0+store.muscleSets($1)}}}
    var weekLabel:String {let interval=Calendar.current.dateInterval(of:.weekOfYear,for:Date())!;return interval.start.formatted(.dateTime.day().month(.abbreviated))+" – "+interval.end.addingTimeInterval(-1).formatted(.dateTime.day().month(.abbreviated))}
    var body:some View {
        ScrollView {
            VStack(alignment:.leading,spacing:22) {
                if store.data.dismissedTips?.contains("goals") != true {
                    VStack(alignment:.leading,spacing:9) {HStack{Text("Mới: mục tiêu set").font(.system(size:17,weight:.semibold));Spacer();Button{store.dismissTip("goals")}label:{Image(systemName:"xmark").foregroundStyle(Dopa.muted).frame(width:32,height:32)}};Text("Mỗi tuần, mỗi nhóm cơ có một số set cần đạt, tính từ mục tiêu, số buổi mỗi tuần và độ dài buổi tập của bạn.").font(.system(size:15)).foregroundStyle(Dopa.muted);Button("Tìm hiểu thêm"){help=true}.font(.system(size:14,weight:.medium)).foregroundStyle(Dopa.lime)}.padding(16).background(Dopa.card).clipShape(RoundedRectangle(cornerRadius:14))
                }
                VStack(alignment:.leading,spacing:5){Text("Mục tiêu set tuần").dopaTitle(29);Text(weekLabel).font(.system(size:13)).foregroundStyle(Dopa.muted)}
                ZStack {
                    ForEach(groups.indices,id:\.self){i in
                        let value=groups[i].muscles.reduce(0){$0+store.muscleSets($1)}
                        Circle().trim(from:CGFloat(i)/3+0.012,to:CGFloat(i+1)/3-0.012).stroke(groups[i].color.opacity(0.3),style:StrokeStyle(lineWidth:14,lineCap:.round)).rotationEffect(.degrees(-90))
                        Circle().trim(from:CGFloat(i)/3+0.012,to:CGFloat(i)/3+0.012+CGFloat(min(1,value/groups[i].target))*(1/3-0.024)).stroke(groups[i].color,style:StrokeStyle(lineWidth:14,lineCap:.round)).rotationEffect(.degrees(-90))
                    }
                    VStack(spacing:6){Text("\(Int(total/65*100))%").font(Dopa.metricFont(38));Text("\(AppStore.number(total)) / 65 set").font(.system(size:15)).foregroundStyle(Dopa.muted)}
                }.frame(width:200,height:200).frame(maxWidth:.infinity).padding(.vertical,5)
                ForEach(groups.indices,id:\.self){index in goalCard(index)}
                Text("Lịch sử").dopaTitle(26).padding(.top,8)
                HStack {ForEach((1...4).reversed(),id:\.self){week in
                    let date=Calendar.current.date(byAdding:.weekOfYear,value:-week,to:Date())!
                    let interval=Calendar.current.dateInterval(of:.weekOfYear,for:date)!
                    let value=store.data.history.filter{interval.contains($0.started)}.flatMap(\.exercises).reduce(0.0){sum,ex in sum+Double(ex.sets.filter{$0.completed && !$0.warmup}.count)*Double(ex.exercise.muscles.count)+Double(ex.sets.filter{$0.completed && !$0.warmup}.count)*Double(ex.exercise.secondary.count)*0.5}
                    VStack(spacing:7){SmallGoalRing(value:value/65,color:Dopa.legs,size:48);Text("\(Int(value/65*100))%").font(.system(size:13));Text(date.formatted(.dateTime.day().month())).font(.system(size:10)).foregroundStyle(Dopa.muted)}.frame(maxWidth:.infinity)
                }}.padding(16).background(Dopa.card).clipShape(RoundedRectangle(cornerRadius:14))
            }.padding(16)
        }.background(Dopa.background).sheet(isPresented:$help){ArticleSheet(title:"Mục tiêu set tuần",sections:[("Cách tính","Mỗi set chính đóng góp 1 set cho cơ chính và 0,5 set cho cơ hỗ trợ. Set khởi động không tính vào mục tiêu."),("Theo dõi tiến độ","Ghi các set đã tập để cập nhật tiến độ. Số set còn lại hiển thị ở từng nhóm cơ.")])}
    }
    func goalCard(_ index:Int)->some View {
        let group=groups[index];let value=group.muscles.reduce(0){$0+store.muscleSets($1)}
        return VStack(spacing:15) {
            Button{if expanded.contains(index){expanded.remove(index)}else{expanded.insert(index)}}label:{HStack(spacing:12){SmallGoalRing(value:value/group.target,color:group.color,size:32);VStack(alignment:.leading,spacing:4){Text(group.title).font(.system(size:18,weight:.semibold));Text("\(AppStore.number(value)) / \(AppStore.number(group.target)) set").font(.system(size:13)).foregroundStyle(Dopa.muted)};Spacer();VStack(alignment:.trailing,spacing:3){Text(AppStore.number(max(0,group.target-value))).font(.system(size:16,weight:.semibold));Text("còn lại").font(.system(size:11)).foregroundStyle(Dopa.muted)};Image(systemName:expanded.contains(index) ? "chevron.down":"chevron.right").font(.caption).foregroundStyle(Dopa.muted)}}.buttonStyle(.plain)
            if expanded.contains(index) {
                Divider()
                ForEach(group.muscles.indices,id:\.self){i in let muscle=group.muscles[i];let count=store.muscleSets(muscle);HStack(spacing:12){SmallGoalRing(value:count/group.targets[i],color:group.color,size:28);VStack(alignment:.leading,spacing:4){Text(muscle).font(.system(size:16));Text("\(AppStore.number(count)) / \(AppStore.number(group.targets[i])) set").font(.system(size:12)).foregroundStyle(Dopa.muted)};Spacer();VStack(alignment:.trailing,spacing:3){Text(AppStore.number(max(0,group.targets[i]-count)));Text("còn lại").font(.system(size:10)).foregroundStyle(Dopa.muted)}}}
                Divider();Text("Cơ chính = 1 set\nCơ phụ = 0,5 set").font(.system(size:12)).foregroundStyle(Dopa.muted).frame(maxWidth:.infinity,alignment:.leading)
            }
        }.padding(16).background(Dopa.card).clipShape(RoundedRectangle(cornerRadius:14))
    }
}
struct SmallGoalRing:View {
    var value:Double
    var color:Color
    var size:CGFloat
    var body:some View {ZStack{Circle().stroke(color.opacity(0.25),lineWidth:5);Circle().trim(from:0,to:min(1,max(0,value))).stroke(color,style:StrokeStyle(lineWidth:5,lineCap:.round)).rotationEffect(.degrees(-90))}.frame(width:size,height:size)}
}
