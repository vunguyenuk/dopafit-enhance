/// Baked CC0 muscular MakeHuman mesh derived with the OpenGym3D generation stack. Regions are approximate
/// bone territories adapted from upstream MUSCLE_SPEC, not Dopafit anatomy masks.
private struct OpenGymMesh {
    var positions:[SCNVector3]
    var normals:[SCNVector3]
    var regionNames:[String]
    var regionWeights:[[Double]]
    var elements:SCNGeometryElement
    var modelHeight:Float
    var formStyle:Bool
    static let shared:OpenGymMesh? = {
        let formURL=Bundle.main.url(forResource:"neutral",withExtension:"glb",subdirectory:"Form3D")
        guard let url=formURL ?? Bundle.main.url(forResource:"muscular_body",withExtension:"glb",subdirectory:"OpenGym3D/models"),
              let data=try? Data(contentsOf:url),data.count>20 else{return nil}
        func u32(_ offset:Int)->Int {Int(data.withUnsafeBytes{$0.loadUnaligned(fromByteOffset:offset,as:UInt32.self)})}
        guard u32(0)==0x46546c67,20+u32(12)<=data.count,
              let root=(try? JSONSerialization.jsonObject(with:data.subdata(in:20..<20+u32(12)))) as? [String:Any],
              let accessors=root["accessors"] as? [[String:Any]],let views=root["bufferViews"] as? [[String:Any]],
              let meshes=root["meshes"] as? [[String:Any]],let primitives=meshes.first?["primitives"] as? [[String:Any]],
              let primitive=primitives.first,let attrs=primitive["attributes"] as? [String:Int],
              let extras=root["extras"] as? [String:Any],let regionNames=extras["openGymMuscleNames"] as? [String],regionNames.count==16 else{return nil}
        let binStart=20+u32(12)+8
        func values(_ index:Int,components:Int)->[[Double]] {
            guard accessors.indices.contains(index) else{return []}
            let a=accessors[index];guard let vi=a["bufferView"] as? Int,views.indices.contains(vi) else{return []}
            let v=views[vi],ct=a["componentType"] as? Int ?? 5126
            let bytes=ct==5126 || ct==5125 ? 4:ct==5123 ? 2:1
            let offset=binStart+(v["byteOffset"] as? Int ?? 0)+(a["byteOffset"] as? Int ?? 0)
            let stride=v["byteStride"] as? Int ?? bytes*components
            let count=a["count"] as? Int ?? 0
            guard count>0,offset+(count-1)*stride+bytes*components<=data.count else{return []}
            return (0..<count).map { i in (0..<components).map {c in
                let p=offset+i*stride+c*bytes
                return data.withUnsafeBytes { raw in
                    switch ct {
                    case 5126:return Double(raw.loadUnaligned(fromByteOffset:p,as:Float.self))
                    case 5125:return Double(raw.loadUnaligned(fromByteOffset:p,as:UInt32.self))
                    case 5123:return Double(raw.loadUnaligned(fromByteOffset:p,as:UInt16.self))
                    default:return Double(raw.load(fromByteOffset:p,as:UInt8.self))
                    }
                }
            }}
        }
        guard let pi=attrs["POSITION"],let ni=attrs["NORMAL"],let ii=primitive["indices"] as? Int else{return nil}
        let p=values(pi,components:3),n=values(ni,components:3)
        guard !p.isEmpty,p.count==n.count else{return nil}
        var maskSources=[[[Double]]]()
        for slot in 0..<4 {
            guard let index=attrs["_MUSCLE_\(slot)"] else{return nil}
            let source=values(index,components:4);guard source.count==p.count else{return nil};maskSources.append(source)
        }
        let masks=p.indices.map{i in maskSources.flatMap{$0[i]}}
        let indices=values(ii,components:1).map{UInt32($0[0])}
        guard indices.allSatisfy({Int($0)<p.count}),indices.count%3==0 else{return nil}
        return OpenGymMesh(positions:p.map{SCNVector3($0[0],$0[1],$0[2])},normals:n.map{SCNVector3($0[0],$0[1],$0[2])},regionNames:regionNames,regionWeights:masks,elements:SCNGeometryElement(indices:indices,primitiveType:.triangles),modelHeight:Float(p.map{$0[1]}.max() ?? 1.8),formStyle:formURL != nil)
    }()
    func geometry(heat:[String:Double])->SCNGeometry {
        // Explicit per-face materials keep SceneKit's PBR path independent of
        // vertex-colour semantics, which differ from glTF's material pipeline.
        let activation=positions.indices.map {i in min(1,max(0,regionNames.indices.map{k in regionWeights[i][k]*(heat[regionNames[k]] ?? 0)}.max() ?? 0))}
        let raw=elements.data
        let indices:[UInt32]=raw.withUnsafeBytes{p in (0..<(raw.count/4)).map{p.loadUnaligned(fromByteOffset:$0*4,as:UInt32.self)}}
        var buckets=Array(repeating:[UInt32](),count:42)
        for i in stride(from:0,to:indices.count-2,by:3) {
            let tri=Array(indices[i...i+2]);let level=Int((tri.reduce(0.0){$0+activation[Int($1)]}/3*20).rounded())
            let y=tri.reduce(Float(0)){$0+positions[Int($1)].y}/3
            let x=tri.reduce(Float(0)){$0+abs(positions[Int($1)].x)}/3
            let cloth=formStyle && y>0.775 && y<1.065 && x<0.32
            buckets[level+(cloth ? 21:0)].append(contentsOf:tri)
        }
        var geometryElements:[SCNGeometryElement]=[],materials:[SCNMaterial]=[]
        for level in buckets.indices where !buckets[level].isEmpty {
            let element=SCNGeometryElement(indices:buckets[level],primitiveType:.triangles);geometryElements.append(element)
            let t=CGFloat(level%21)/20,material=SCNMaterial();material.lightingModel = .physicallyBased
            // SceneKit's PBR lights otherwise clip sRGB orange to yellow.
            // Compensated base colour retains the intended orange on-device.
            material.diffuse.contents=UIColor(red:0.12+(0.85-0.12)*t,green:0.15+(0.21-0.15)*t,blue:0.19+(0.015-0.19)*t,alpha:1)
            if formStyle {
                let base=level>=21 ? SIMD3<Double>(0.204,0.231,0.212):SIMD3<Double>(0.57,0.60,0.58)
                material.diffuse.contents=UIColor(red:base.x+(0.85-base.x)*Double(t),green:base.y+(0.21-base.y)*Double(t),blue:base.z+(0.035-base.z)*Double(t),alpha:1)
            }
            material.name=String(Double(t));material.metalness.contents=formStyle ? 0:0.15;material.roughness.contents=formStyle ? 0.66:0.7
            material.emission.contents=UIColor(red:1,green:0.3,blue:0.015,alpha:1);material.emission.intensity=t*0.04;materials.append(material)
        }
        let geometry=SCNGeometry(sources:[SCNGeometrySource(vertices:positions),SCNGeometrySource(normals:normals)],elements:geometryElements);geometry.materials=materials;return geometry
    }
}

enum OpenGymBodyScene {
    static func scene(heat:[String:Double],rear:Bool=false,focus:String?=nil,angle:Int=0)->SCNScene {
        let scene=SCNScene();scene.background.contents=UIColor.clear
        if let mesh=OpenGymMesh.shared {
            let body=SCNNode(geometry:mesh.geometry(heat:heat));body.name="body";scene.rootNode.addChildNode(body)
        }
        let camera=SCNNode();camera.name="camera";camera.camera=SCNCamera();camera.camera?.usesOrthographicProjection=true
        let backMuscles=["Lưng","Lưng dưới","Mông","Đùi sau","Tay sau","Cơ thang","Bắp chân"]
        let isRear=rear || focus.map{backMuscles.contains($0)} == true
        let height=OpenGymMesh.shared?.modelHeight ?? 1.8
        var center:Float=height/2;var scale=Double(height)*0.68
        if let focus {
            switch focus {
            case "Ngực","Lưng":center=1.32;scale=0.43
            case "Bụng","Lưng dưới":center=1.1;scale=0.4
            case "Mông","Cơ dạng hông","Cơ khép hông":center=0.87;scale=0.39
            case "Đùi trước","Đùi sau":center=0.66;scale=0.45
            case "Bắp chân":center=0.31;scale=0.35
            case "Cổ","Cơ thang":center=1.45;scale=0.35
            default:center=1.16;scale=0.52
            }
        }
        if focus != nil {center *= height/1.8}
        let yaw=isRear ? Float.pi:Float(angle)*Float.pi/2
        camera.camera?.orthographicScale=scale;camera.position=SCNVector3(sin(yaw)*3,center,cos(yaw)*3);camera.eulerAngles.y=yaw
        camera.camera?.zNear=0.05;camera.camera?.zFar=10;scene.rootNode.addChildNode(camera)
        for (position,intensity) in [(SCNVector3(2,3,4),CGFloat(350)),(SCNVector3(-2,1,-3),CGFloat(180))] {
            let light=SCNNode();light.light=SCNLight();light.light?.type = .omni;light.light?.intensity=intensity;light.position=position;scene.rootNode.addChildNode(light)
        }
        let ambient=SCNNode();ambient.light=SCNLight();ambient.light?.type = .ambient;ambient.light?.intensity=100;scene.rootNode.addChildNode(ambient)
        return scene
    }
    static func setPulse(_ scene:SCNScene?,enabled:Bool) {
        guard let body=scene?.rootNode.childNode(withName:"body",recursively:false) else{return}
        body.removeAction(forKey:"musclePulse")
        let materials=(body.geometry?.materials ?? []).map{($0,CGFloat(Double($0.name ?? "0") ?? 0))}
        for (material,level) in materials{material.emission.intensity=level*0.04;material.diffuse.intensity=1}
        guard enabled,materials.contains(where:{$0.1>0}) else{return}
        let action=SCNAction.customAction(duration:1.8){_,elapsed in
            let wave=0.5-0.5*cos(Double(elapsed)/1.8*2*Double.pi)
            for (material,level) in materials{
                material.emission.intensity=level*CGFloat(0.02+0.15*wave)
                material.diffuse.intensity=1-level*CGFloat(0.4*(1-wave))
            }
        }
        body.runAction(.repeatForever(action),forKey:"musclePulse")
    }
    static let cache=NSCache<NSString,UIImage>()
    static func thumbnail(_ muscle:String)->UIImage {
        if let image=cache.object(forKey:muscle as NSString){return image}
        let scene=scene(heat:[muscle:1],focus:muscle),renderer=SCNRenderer(device:nil,options:nil);renderer.scene=scene;renderer.pointOfView=scene.rootNode.childNode(withName:"camera",recursively:false)
        let image=renderer.snapshot(atTime:0,with:CGSize(width:192,height:192),antialiasingMode:.multisampling4X)
        cache.setObject(image,forKey:muscle as NSString);return image
    }
}
struct OpenGymBodyView:UIViewRepresentable {
    var heat:[String:Double]=[:]
    var rear=false
    var angle=0
    var pulse=false
    @Environment(\.accessibilityReduceMotion) private var reduceMotion
    @Environment(\.scenePhase) private var phase
    final class Coordinator {var key="";var pulsing=false}
    func makeCoordinator()->Coordinator{Coordinator()}
    func makeUIView(context:Context)->SCNView {let view=SCNView();view.backgroundColor = .clear;view.autoenablesDefaultLighting=false;view.allowsCameraControl=true;view.preferredFramesPerSecond=30;return view}
    func updateUIView(_ view:SCNView,context:Context) {
        let key=heat.keys.sorted().map{"\($0):\(Int((heat[$0] ?? 0)*100))"}.joined()+String(rear)+String(angle)
        let changed=context.coordinator.key != key
        context.coordinator.key=key
        let pulsing=pulse && !reduceMotion && phase == .active
        view.accessibilityValue=heat.keys.filter{(heat[$0] ?? 0)>0}.sorted().joined(separator:", ")
        if changed{view.scene=OpenGymBodyScene.scene(heat:heat,rear:rear,angle:angle);view.pointOfView=view.scene?.rootNode.childNode(withName:"camera",recursively:false)}
        if changed || pulsing != context.coordinator.pulsing{OpenGymBodyScene.setPulse(view.scene,enabled:pulsing)}
        context.coordinator.pulsing=pulsing;view.isPlaying=pulsing
        view.accessibilityLabel=rear ? "Mô hình cơ thể mặt sau":"Mô hình cơ thể mặt trước"
    }
}
struct OpenGymMuscleImage:View {
    var muscle:String
    var body:some View {Image(uiImage:OpenGymBodyScene.thumbnail(muscle)).resizable().scaledToFit().accessibilityHidden(true)}
}
enum OpenGymMedia {
    static let mapping=["dumbbell-curl":"form:standing-dumbbell-curl","rdl":"form:romanian-deadlift","squat":"form:barbell-back-squat","goodmorning":"good_morning","incline-dumbbell-curl":"form:incline-dumbbell-curl","dumbbell-lateral-raise":"form:dumbbell-lateral-raise","dumbbell-shoulder-press":"form:dumbbell-shoulder-press","goblet-squat":"form:goblet-squat","bent-over-dumbbell-row":"form:bent-over-dumbbell-row"]
    static func model(for exercise:Exercise)->String?{mapping[exercise.id]}
    static func heat(for exercise:Exercise)->[String:Double] {
        var heat=Dictionary(uniqueKeysWithValues:exercise.secondary.map{($0,0.45)})
        for muscle in exercise.muscles{heat[muscle]=1};return heat
    }
}
